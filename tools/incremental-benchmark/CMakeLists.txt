cmake_minimum_required(VERSION 3.28)
project(VelocityCopyIncrementalBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT VELOCITYCOPY_REPO_ROOT)
    message(FATAL_ERROR "Set VELOCITYCOPY_REPO_ROOT to the exact checkout to measure")
endif()
add_subdirectory("${VELOCITYCOPY_REPO_ROOT}" velocitycopy)
add_executable(VelocityCopyIncrementalBenchmark benchmark.cpp)
target_link_libraries(VelocityCopyIncrementalBenchmark PRIVATE velocitycopy_core psapi)
target_compile_options(VelocityCopyIncrementalBenchmark PRIVATE /W4 /permissive- /EHsc /utf-8)
